import os
import json
import logging
from datetime import datetime
from .ftp_client import FTPClient
from .ym_client import YandexMarketClient

logger = logging.getLogger(__name__)

class StockUpdater:
    def __init__(self):
        self.ftp = FTPClient(
            host=os.environ['FTP_HOST'],
            user=os.environ['FTP_USER'],
            password=os.environ['FTP_PASS']
        )
        self.ym = YandexMarketClient(
            api_key=os.environ['YM_API_KEY'],
            campaign_id=os.environ['YM_CAMPAIGN_ID'],
            warehouse_id=os.environ['YM_WAREHOUSE_ID'],
            business_id=os.environ['YM_BUSINESS_ID']
        )

    def run(self):
        try:
            print("🔄 Обновляем остатки...")
            os.makedirs("src/data/stock", exist_ok=True)
            files = self.ftp.list_files(os.environ.get('FTP_FOLDER', '/'), 'stock')
            
            if not files:
                print("⚠️ Нет файлов")
                return False
            
            stocks = []
            for f in files:
                local = f"src/data/stock/{os.path.basename(f)}"
                if self.ftp.download_file(f, local):
                    with open(local, 'r') as file:
                        data = json.load(file)
                        stocks.extend(data)
            
            if stocks:
                response = self.ym.update_stock(stocks)
                result = response.json() if response else {}
                
                # Сохраняем результат
                with open(f"src/data/stock/update_{datetime.now().strftime('%Y%m%d_%H%M%S')}.json", 'w') as f:
                    json.dump({"stocks": stocks, "result": result}, f, indent=2)
                print(f"✅ Обновлено {len(stocks)} товаров")
                return True
            return False
        except Exception as e:
            print(f"❌ Ошибка: {e}")
            return False
