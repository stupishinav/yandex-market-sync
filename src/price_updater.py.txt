import os
import json
import logging
from datetime import datetime
from .ftp_client import FTPClient
from .ym_client import YandexMarketClient

logger = logging.getLogger(__name__)

class PriceUpdater:
    def __init__(self):
        self.ftp = FTPClient(
            host=os.environ['FTP_HOST'],
            user=os.environ['FTP_USER'],
            password=os.environ['FTP_PASS']
        )
        self.ym = YandexMarketClient(
            api_key=os.environ['YM_API_KEY'],
            campaign_id=os.environ['YM_CAMPAIGN_ID'],
            warehouse_id=os.environ['YM_WAREHOUSE_ID'],
            business_id=os.environ['YM_BUSINESS_ID']
        )

    def run(self):
        try:
            print("🔄 Обновляем цены...")
            os.makedirs("src/data/prices", exist_ok=True)
            price_folder = os.environ.get('FTP_PRICE_FOLDER', 'prices/')
            files = self.ftp.list_files(price_folder, 'price')
            
            if not files:
                print("⚠️ Нет файлов с ценами")
                return False
            
            prices = []
            for f in files:
                local = f"src/data/prices/{os.path.basename(f)}"
                if self.ftp.download_file(f, local):
                    with open(local, 'r') as file:
                        data = json.load(file)
                        if isinstance(data, list):
                            prices.extend(data)
            
            if prices:
                payload = {
                    "prices": [
                        {"id": p['offer_id'], "price": {"value": str(p['price']), "currencyId": "RUR"}}
                        for p in prices
                    ]
                }
                # Добавляем старую цену
                for p, item in zip(prices, payload['prices']):
                    if 'old_price' in p:
                        item['price']['discountBase'] = str(p['old_price'])
                
                response = self.ym.update_prices(payload)
                result = response.json() if response else {}
                
                with open(f"src/data/prices/update_{datetime.now().strftime('%Y%m%d_%H%M%S')}.json", 'w') as f:
                    json.dump({"prices": prices, "result": result}, f, indent=2)
                print(f"✅ Обновлено {len(prices)} цен")
                return True
            return False
        except Exception as e:
            print(f"❌ Ошибка: {e}")
            return False
